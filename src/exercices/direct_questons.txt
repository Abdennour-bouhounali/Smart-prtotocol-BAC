\section*{المحور الأول: الدوال العددية، الأسية واللوغاريتمية}
\begin{enumerate}
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{x^3}{\ln x - x}$ معرفة على $(1, +\infty)$. \\
    \textbf{السؤال:} احسب النهاية: $\lim_{x \to +\infty} f(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{e^{2x} - e^x}{x}$ معرفة على $\mathbb{R}^*$. \\
    \textbf{السؤال:} باستخدام تعريف العدد المشتق، احسب النهاية: $\lim_{x \to 0} f(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \sqrt{x^2+x+1}$ ولدينا المستقيم المقارب المائل $\Delta: y = ax + b$ بجوار $+\infty$. \\
    \textbf{السؤال:} عين العددين الحقيقيين $a$ و $b$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x^2 (\ln x)^3$ معرفة على $(0, +\infty)$. \\
    \textbf{السؤال:} احسب النهاية: $\lim_{x \to 0^>} f(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x \ln\left(1 + \frac{1}{x}\right)$ معرفة على $(0, +\infty)$. \\
    \textbf{السؤال:} احسب النهاية: $\lim_{x \to +\infty} f(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \ln(x + \sqrt{x^2+1})$ المعرفة على $\mathbb{R}$. \\
    \textbf{السؤال:} عين الدالة المشتقة $f'(x)$ وبسط العبارة إلى أقصى حد.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x^{x}$ معرفة على $(0, +\infty)$. \\
    \textbf{السؤال:} عين الدالة المشتقة $f'(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = (x+1)e^{-x}$ المعرفة على $\mathbb{R}$. \\
    \textbf{السؤال:} برهن أن الدالة $f$ تحقق المعادلة التفاضلية $y'' + 2y' + y = 0$.
    
    \item \textbf{المعطيات:} لتكن الدالة المساعدة $g(x) = x - 1 - \ln x$ المعرفة على $(0, +\infty)$. \\
    \textbf{السؤال:} ادرس اتجاه تغير الدالة $g$ ثم استنتج إشارتها على مجال تعريفها.
    
    \item \textbf{المعطيات:} لتكن $g(x) = x^2 + 1 - \ln x$ حيث $g(\alpha) = 0$، ولتكن الدالة $f(x) = \frac{x + \ln x}{x-1}$. \\
    \textbf{السؤال:} برهن أن $f(\alpha) = \frac{\alpha+1}{\alpha-1}$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x^2 e^{x-1}$ و $(C_f)$ منحناها البياني. \\
    \textbf{السؤال:} ناقش بيانياً وحسب قيم الوسيط الحقيقي $m$ عدد وإشارة حلول المعادلة $x^2 e^{x-1} = m$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{x^2+1}{x}$ و $(C_f)$ منحناها البياني، و $m$ وسيط حقيقي غير معدوم. \\
    \textbf{السؤال:} ناقش بيانياً عدد حلول المعادلة $f(x) = f(m)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{x^2+3}{x+1}$ و $(C_f)$ منحناها البياني، ولتكن النقطة $A(0, -1)$ التي لا تنتمي للمنحنى. \\
    \textbf{السؤال:} اكتب معادلات المماسات للمنحنى $(C_f)$ التي تشمل النقطة $A$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{2x^2-4x+5}{x-1}$ ولتكن النقطة $\Omega(1, 0)$. \\
    \textbf{السؤال:} برهن أن النقطة $\Omega$ هي مركز تناظر للمنحنى $(C_f)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x + 1 - \frac{x}{e^x-1}$ والمستقيم $\Delta: y = x+1$. \\
    \textbf{السؤال:} ادرس وضعية المنحنى $(C_f)$ بالنسبة للمقارب المائل $\Delta$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x e^{1-x}$. \\
    \textbf{السؤال:} برهن أنه من أجل كل $x \in [0, 1]$، لدينا المتراجحة $f(x) \le x$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f$ المعرفة على $\mathbb{R}$ حيث $f'(x) = x(x-2)$، ولتكن الدالة المركبة $h(x) = f(x^2)$. \\
    \textbf{السؤال:} استنتج اتجاه تغير الدالة $h$ على $\mathbb{R}$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = x^3 + x - 1$. \\
    \textbf{السؤال:} بين أن المعادلة $f(x) = 0$ تقبل حلاً وحيداً $\alpha$ حيث $0.6 < \alpha < 0.7$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = e^x + x$ المعرفة على $\mathbb{R}$ والتي تقبل دالة عكسية $f^{-1}$. \\
    \textbf{السؤال:} احسب العدد المشتق $(f^{-1})'(1)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \ln|x|$ و $(C_f)$ منحناها، ولتكن الدالة $k(x) = |f(|x|)|$. \\
    \textbf{السؤال:} اشرح كيف يمكن رسم منحنى الدالة $k$ انطلاقاً من منحنى الدالة $f$.
\end{enumerate}

\section*{المحور الثاني: المتتاليات العددية}
\begin{enumerate}
    \item \textbf{المعطيات:} لتكن المتتالية $(u_n)$ المعرفة بـ $u_0 = \frac{1}{2}$ و $u_{n+1} = \frac{2u_n}{u_n+1}$. \\
    \textbf{السؤال:} برهن بالتراجع أنه من أجل كل عدد طبيعي $n$: $0 < u_n < 1$.
    
    \item \textbf{المعطيات:} لتكن المتتالية $(u_n)$ حيث $u_{n+1} - u_n = \frac{-u_n^2 + u_n + 2}{u_n + 3}$ ونعلم أن $0 < u_n < 2$. \\
    \textbf{السؤال:} ادرس اتجاه تغير المتتالية $(u_n)$.
    
    \item \textbf{المعطيات:} لتكن المتتالية المعرفة بـ $u_0 = 1$ و $u_{n+1} = \sqrt{u_n + 2}$. \\
    \textbf{السؤال:} برهن بالتراجع أن المتتالية $(u_n)$ متزايدة ومحدودة من الأعلى بالعدد 2.
    
    \item \textbf{المعطيات:} لتكن المتتالية الموجبة $(u_n)$ حيث يعطى من أجل كل $n \in \mathbb{N}$ أن $u_{n+1} \le \frac{3}{4} u_n$ و $u_0 = 1$. \\
    \textbf{السؤال:} برهن بالتراجع أن $0 < u_n \le \left(\frac{3}{4}\right)^n$.
    
    \item \textbf{المعطيات:} لدينا من السؤال السابق الحصر $0 < u_n \le \left(\frac{3}{4}\right)^n$. \\
    \textbf{السؤال:} استنتج باستخدام مبرهنة الحصر نهاية المتتالية $(u_n)$.
    
    \item \textbf{المعطيات:} لتكن $(u_n)$ المعرفة بـ $u_0 = 0$ و $u_{n+1} = \frac{2u_n+3}{u_n+4}$، ولتكن المتتالية $v_n = \frac{u_n - \alpha}{u_n + 1}$. \\
    \textbf{السؤال:} عين العدد الحقيقي $\alpha$ حتى تكون المتتالية $(v_n)$ هندسية.
    
    \item \textbf{المعطيات:} لتكن $(v_n)$ متتالية هندسية أساسها $q=\frac{1}{2}$ وحدها الأول $v_0=3$ ولدينا المجموع $S_n = v_0 + v_1 + \dots + v_n$. \\
    \textbf{السؤال:} احسب $S_n$ بدلالة $n$.
    
    \item \textbf{المعطيات:} انطلاقاً من المعطيات السابقة، لتكن العلاقة $u_n = \frac{v_n}{v_n-1}$ ولدينا المجموع $S'_n = \frac{1}{v_0-1} + \dots + \frac{1}{v_n-1}$. \\
    \textbf{السؤال:} استنتج المجموع $S'_n$ بدلالة $n$ والمجموع السابق $S_n$.
    
    \item \textbf{المعطيات:} لتكن المتتالية الحسابية $(v_n)$ أساسها $r=2$ وحدها الأول $v_0=1$ ولدينا الجداء $P_n = 2^{v_0} \times 2^{v_1} \times \dots \times 2^{v_n}$. \\
    \textbf{السؤال:} احسب الجداء $P_n$ بدلالة $n$.
    
    \item \textbf{المعطيات:} لتكن المتتالية الهندسية $(v_n)$ الموجبة تماماً ولدينا المجموع $T_n = \ln(v_0) + \ln(v_1) + \dots + \ln(v_n)$. \\
    \textbf{السؤال:} احسب المجموع $T_n$ بدلالة $n$.
    
    \item \textbf{المعطيات:} لتكن المتتالية التكاملية $I_n = \int_0^1 x^n e^x dx$ المعرفة من أجل كل $n \ge 1$. \\
    \textbf{السؤال:} باستخدام المكاملة بالتجزئة، برهن العلاقة التراجعية: $I_{n+1} = e - (n+1)I_n$.
    
    \item \textbf{المعطيات:} لتكن المتتالية السابقة $I_n = \int_0^1 x^n e^x dx$. \\
    \textbf{السؤال:} برهن أن المتتالية $(I_n)$ متناقصة واستنتج أنها متقاربة.
    
    \item \textbf{المعطيات:} ليكن المتغير $x$ بحيث $x \in [0, 1]$. \\
    \textbf{السؤال:} برهن أنه من أجل كل عدد طبيعي غير معدوم $n$، لدينا: $0 \le x^n e^x \le x^n e$.
    
    \item \textbf{المعطيات:} بناءً على الحصر السابق $0 \le x^n e^x \le x^n e$. \\
    \textbf{السؤال:} استنتج أن $0 \le I_n \le \frac{e}{n+1}$، ثم احسب $\lim I_n$.
    
    \item \textbf{المعطيات:} لتكن المتتاليتان $u_{n+1} = \frac{u_n + v_n}{2}$ و $v_{n+1} = \frac{u_n + 2v_n}{3}$ مع $u_0 = 1, v_0 = 2$. \\
    \textbf{السؤال:} برهن أن المتتاليتين $(u_n)$ و $(v_n)$ متجاورتان.
    
    \item \textbf{المعطيات:} لتكن المتتاليتان السابقتان $(u_n)$ و $(v_n)$ ولتكن المتتالية المساعدة $w_n = a u_n + b v_n$. \\
    \textbf{السؤال:} عين الثابتين الحقيقيين $a$ و $b$ غير المعدومين بحيث تكون المتتالية $(w_n)$ ثابتة.
    
    \item \textbf{المعطيات:} ليكن المجموع $S_n = \sum_{k=0}^n k \times q^k$ حيث $q$ أساس متتالية هندسية ($q \neq 1$). \\
    \textbf{السؤال:} احسب المجموع $S_n$ بدلالة $n$ و $q$.
    
    \item \textbf{المعطيات:} لتكن المتتالية المعرفة بالمجموع $u_n = \sum_{k=1}^n \frac{1}{k!}$. \\
    \textbf{السؤال:} أثبت أن $(u_n)$ متزايدة ومحدودة من الأعلى بالعدد 3، ثم استنتج تقاربها.
    
    \item \textbf{المعطيات:} لتكن المتتالية المعرفة بـ $u_0 = 0, u_1 = 1$ والعلاقة التراجعية $u_{n+2} = 3u_{n+1} - 2u_n$. \\
    \textbf{السؤال:} اكتب عبارة الحد العام $u_n$ بدلالة $n$ (باستعمال متتالية مساعدة $v_n = u_{n+1} - u_n$).
    
    \item \textbf{المعطيات:} ليكن المجموع $S_n = \ln\left(1-\frac{1}{2^2}\right) + \ln\left(1-\frac{1}{3^2}\right) + \dots + \ln\left(1-\frac{1}{(n+1)^2}\right)$. \\
    \textbf{السؤال:} بسط العبارة مستعملاً خواص اللوغاريتم ثم ادرس نهاية المجموع $S_n$ عندما يؤول $n$ إلى $+\infty$.
\end{enumerate}

\section*{المحور الثالث: الأعداد المركبة والتحويلات النقطية}
\begin{enumerate}
    \item \textbf{المعطيات:} لتكن المعادلة $z^2 - 2\cos(\theta)z + 1 = 0$ حيث $\theta \in ]0, \pi[$. \\
    \textbf{السؤال:} حل في مجموعة الأعداد المركبة $\mathbb{C}$ هذه المعادلة.
    
    \item \textbf{المعطيات:} لتكن المعادلة $(E): z^3 - (1+i)z^2 + a z + b = 0$ حيث $a, b \in \mathbb{R}$، ونعلم أنها تقبل حلاً تخيلياً صرفاً $z_0 = 2i$. \\
    \textbf{السؤال:} عين العددين $a$ و $b$ ثم حل المعادلة $(E)$.
    
    \item \textbf{المعطيات:} ليكن العدد المركب $Z = \left(\frac{\sqrt{3}+i}{1-i}\right)^{12}$. \\
    \textbf{السؤال:} اكتب العدد $Z$ على الشكل الجبري.
    
    \item \textbf{المعطيات:} ليكن العدد المركب $Z = 1 + e^{i\theta}$ حيث $\theta \in ]-\pi, \pi[$. \\
    \textbf{السؤال:} اكتب العدد $Z$ على الشكل الأسي (باستعمال زاوية النصف) واستنتج طويلته وعمدته.
    
    \item \textbf{المعطيات:} لتكن النقط $A, B, C$ لواحقها $z_A=1+i, z_B=2+2i, z_C=3+i$. \\
    \textbf{السؤال:} احسب النسبة $\frac{z_A - z_B}{z_C - z_B}$ ثم استنتج طبيعة المثلث $ABC$.
    
    \item \textbf{المعطيات:} لتكن النقطتان $A(2i)$ و $B(-1+i)$. \\
    \textbf{السؤال:} عين طبيعة مجموعة النقط $M(z)$ بحيث: $|z - 2i| = |z + 1 - i|$.
    
    \item \textbf{المعطيات:} لتكن النقطة $A(1+i)$. \\
    \textbf{السؤال:} عين طبيعة مجموعة النقط $M(z)$ بحيث: $|(1+i)z - 2i| = 2$.
    
    \item \textbf{المعطيات:} لتكن النقطة $A(i)$. \\
    \textbf{السؤال:} عين طبيعة مجموعة النقط $M(z)$ بحيث: $\arg(z - i) = \frac{\pi}{4} + 2k\pi$.
    
    \item \textbf{المعطيات:} لتكن النقطتان $A(z_A)$ و $B(z_B)$. \\
    \textbf{السؤال:} عين طبيعة مجموعة النقط $M(z)$ بحيث يكون العدد $\frac{z-z_A}{z-z_B}$ تخيلياً صرفاً.
    
    \item \textbf{المعطيات:} لتكن النقط $A(1), B(i), C(1+i)$. \\
    \textbf{السؤال:} عين الكتابة المركبة للتشابه المباشر $S$ الذي مركزه $A$ ويحول النقطة $B$ إلى النقطة $C$.
    
    \item \textbf{المعطيات:} ليكن التحويل النقطي المعرف بعبارته المركبة $z' = (1+i)z + 2 - i$. \\
    \textbf{السؤال:} عين طبيعة هذا التحويل النقطي وعناصره المميزة.
    
    \item \textbf{المعطيات:} لتكن الدائرة $(C)$ ذات المركز $\Omega(1, -1)$ ونصف القطر 2، والتشابه المباشر $S$ ذو النسبة $\sqrt{2}$ والزاوية $\frac{\pi}{4}$ والمركز المبدأ $O$. \\
    \textbf{السؤال:} عين معادلة (أو مميزات) الدائرة $(C')$ صورة الدائرة $(C)$ بالتشابه $S$.
    
    \item \textbf{المعطيات:} لتكن النقط $A(1+i), B(2+3i), C(-1-3i)$. \\
    \textbf{السؤال:} باستخدام الأعداد المركبة، برهن أن النقط $A, B, C$ في استقامية.
    
    \item \textbf{المعطيات:} لتكن المعادلة $z = \bar{z} + 2i$. \\
    \textbf{السؤال:} حل في $\mathbb{C}$ هذه المعادلة وعين طبيعة مجموعة النقط صور الحلول.
    
    \item \textbf{المعطيات:} لتكن المعادلة $z^5 = 1$. \\
    \textbf{السؤال:} حل في $\mathbb{C}$ المعادلة واستنتج الجذور الخماسية للوحدة، ثم مثلها في المستوي المركب.
    
    \item \textbf{المعطيات:} ليكن العدد المركب $Z = \sqrt{3} - i$. \\
    \textbf{السؤال:} عين قيم العدد الطبيعي $n$ حتى يكون العدد $Z^n$ حقيقياً موجباً تماماً.
    
    \item \textbf{المعطيات:} ليكن العدد المركب $Z = \frac{z^2 - 1}{z}$ حيث $z \neq 0$. \\
    \textbf{السؤال:} بين أن العدد $Z$ هو عدد حقيقي إذا وفقط إذا كان $|z| = 1$ أو $z$ عدد حقيقي غير معدوم.
    
    \item \textbf{المعطيات:} نعتبر المتتالية المركبة $z_{n+1} = \frac{1+i}{2} z_n$ مع $z_0=2$. \\
    \textbf{السؤال:} اكتب عبارة الطويلة $|z_n|$ بدلالة $n$ ثم احسب نهايتها.
    
    \item \textbf{المعطيات:} انطلاقاً من العلاقة $z_{n+1} = \frac{1+i}{2} z_n$. \\
    \textbf{السؤال:} استنتج طبيعة التحويل النقطي الذي يرفق بكل نقطة $M_n(z_n)$ النقطة $M_{n+1}(z_{n+1})$ وعين عناصره المميزة.
    
    \item \textbf{المعطيات:} لتكن الجملة النقطية $\{(A, 1), (B, -2), (C, 2)\}$ ولواحقها $z_A=1, z_B=i, z_C=-i$. \\
    \textbf{السؤال:} عين اللاحقة $z_G$ لمرجح الجملة النقطية $G$.
\end{enumerate}

\section*{المحور الرابع: الدوال الأصلية والحساب المكاملي}
\begin{enumerate}
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{2x+1}{(x^2+x+1)^2}$ المعرفة على $\mathbb{R}$. \\
    \textbf{السؤال:} عين الدالة الأصلية $F$ للدالة $f$ التي تنعدم من أجل $x = 0$.
    
    \item \textbf{المعطيات:} ليكن التكامل المحدّد $I = \int_1^e x^2 \ln x dx$. \\
    \textbf{السؤال:} باستخدام المكاملة بالتجزئة، احسب قيمة $I$.
    
    \item \textbf{المعطيات:} ليكن التكامل $J = \int_0^{\pi/2} e^x \cos x dx$. \\
    \textbf{السؤال:} باستخدام المكاملة بالتجزئة مرتين، احسب قيمة $J$.
    
    \item \textbf{المعطيات:} ليكن المنحنى $(C_f)$ للدالة $f(x)=e^x$ والمستقيم $\Delta: y=x$. \\
    \textbf{السؤال:} احسب مساحة الحيز المستوي المحدد بـ $(C_f)$ و $\Delta$ والمستقيمين $x=0, x=1$.
    
    \item \textbf{المعطيات:} ليكن التكامل المتضمن قيمة مطلقة $K = \int_{-1}^2 |x^2-1| dx$. \\
    \textbf{السؤال:} احسب قيمة التكامل $K$ بفصل المجالات (علاقة شال).
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \sqrt{x e^x}$ المعرفة على المجال $[0, 1]$. \\
    \textbf{السؤال:} احسب الحجم $V$ المولد بدوران المنحنى $(C_f)$ حول محور الفواصل دورة كاملة.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{1}{1+e^x}$ المعرفة على المجال $[0, 1]$. \\
    \textbf{السؤال:} احسب القيمة المتوسطة للدالة $f$ على هذا المجال.
    
    \item \textbf{المعطيات:} لتكن الدالة التكاملية $F(x) = \int_0^x (t^2-1)e^t dt$. \\
    \textbf{السؤال:} ادرس اتجاه تغير الدالة $F$ على $\mathbb{R}$.
    
    \item \textbf{المعطيات:} لتكن المعادلة التفاضلية $(E): y' - 2y = e^x$. \\
    \textbf{السؤال:} عين حلاً خاصاً للمعادلة $(E)$ من الشكل $y_0(x) = a e^x$ ثم استنتج الحل العام للمعادلة.
    
    \item \textbf{المعطيات:} لتكن المعادلة التفاضلية $y'' + 4y = 0$. \\
    \textbf{السؤال:} عين الحل الخاص $f$ الذي يحقق الشروط الابتدائية $f(0)=1$ و $f'(0)=0$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{x^2-x+1}{(x-1)^2}$ المعرفة على $\mathbb{R} \setminus \{1\}$. \\
    \textbf{السؤال:} عين الأعداد الحقيقية $a, b, c$ بحيث $f(x) = a + \frac{b}{x-1} + \frac{c}{(x-1)^2}$ ثم استنتج دالتها الأصلية.
    
    \item \textbf{المعطيات:} ليكن التكامل $I = \int_0^1 e^{x^2} dx$. \\
    \textbf{السؤال:} برهن الحصر التالي: $1 \le I \le e$ (بدون حساب التكامل).
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{1}{x \ln x}$ المعرفة على $(1, +\infty)$. \\
    \textbf{السؤال:} عين مجموعة الدوال الأصلية للدالة $f$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \cos^3 x$ المعرفة على $\mathbb{R}$. \\
    \textbf{السؤال:} استنتج الدالة الأصلية للدالة $f$ باستعمال الدستور $\cos^2 x = 1 - \sin^2 x$.
    
    \item \textbf{المعطيات:} لتكن $f(x) = x - \ln x$ ومماسها $(T)$ عند النقطة ذات الفاصلة 1. \\
    \textbf{السؤال:} احسب مساحة الحيز المحصور بين المنحنى $(C_f)$ والمماس $(T)$ والمستقيمين $x=1$ و $x=e$.
    
    \item \textbf{المعطيات:} ليكن التكامل $I(x) = \int_1^x \frac{1}{t^2} dt$ حيث $x > 1$. \\
    \textbf{السؤال:} احسب $I(x)$ بدلالة $x$ ثم احسب $\lim_{x \to +\infty} I(x)$.
    
    \item \textbf{المعطيات:} لتكن الدالة $f(x) = \frac{e^x - e^{-x}}{e^x + e^{-x}}$. \\
    \textbf{السؤال:} احسب الدالة الأصلية $F(x)$ للدالة $f$ على $\mathbb{R}$.
    
    \item \textbf{المعطيات:} لتكن $f$ دالة فردية ومستمرة على المجال التناظري $[-a, a]$. \\
    \textbf{السؤال:} بين أن $\int_{-a}^a f(x) dx = 0$ باستعمال تغيير المتغير $t = -x$.
    
    \item \textbf{المعطيات:} لتكن الدالتان $f(x) = \ln x$ و $F(x) = x \ln x - x$. \\
    \textbf{السؤال:} برهن أن $F$ هي الدالة الأصلية لـ $f$ على المجال $(0, +\infty)$.
    
    \item \textbf{المعطيات:} لتكن المتتالية $u_n = \frac{1}{n} \sum_{k=1}^n e^{\frac{k}{n}}$. \\
    \textbf{السؤال:} باستخدام تقريب مجاميع ريمان، احسب نهاية المتتالية $\lim_{n \to +\infty} u_n$.
\end{enumerate}

\section*{المحور الخامس: الاحتمالات}
\begin{enumerate}
    \item \textbf{المعطيات:} كيس $A$ يحوي 3 كرات حمراء وكرتين بيضاوين، وكيس $B$ يحوي كرة حمراء و 4 كرات بيضاء. نسحب كرة من $A$ ونضعها في $B$، ثم نسحب كرة من $B$. \\
    \textbf{السؤال:} ما احتمال أن تكون الكرتان المسحوبتان من نفس اللون؟
    
    \item \textbf{المعطيات:} لدينا لعبة ورق تتكون من 32 ورقة. نسحب عشوائياً 3 أوراق في آن واحد. \\
    \textbf{السؤال:} احسب احتمال الحصول على "آس" واحد على الأقل في السحبة.
    
    \item \textbf{المعطيات:} مصنع ينتج قطعاً، 5\% منها معيبة. آلة الفحص ترفض 98\% من القطع المعيبة وترفض 2\% من القطع السليمة. \\
    \textbf{السؤال:} ما احتمال أن تكون القطعة سليمة علماً أنها رُفضت من طرف الآلة؟
    
    \item \textbf{المعطيات:} ثلاثة صناديق: $U_1$ يحوي (كرتين حمراوين، 3 سوداء)، $U_2$ (1 حمراء، 4 سوداء)، $U_3$ (3 حمراء، 2 سوداء). نختار صندوقاً عشوائياً (بنفس الاحتمال) ونسحب منه كرة. \\
    \textbf{السؤال:} استخدم قانون الاحتمال الكلي لحساب احتمال سحب كرة حمراء.
    
    \item \textbf{المعطيات:} كيس يحوي 4 كرات حمراء و 6 بيضاء. نسحب 3 كرات في آن واحد. ليكن $X$ المتغير العشوائي الذي يمثل عدد الكرات الحمراء المسحوبة. \\
    \textbf{السؤال:} عين قانون احتمال المتغير العشوائي $X$.
    
    \item \textbf{المعطيات:} بالاعتماد على المتغير العشوائي $X$ المعرف في السؤال السابق. \\
    \textbf{السؤال:} احسب الأمل الرياضي $E(X)$، والتباين $V(X)$ للمتغير $X$.
    
    \item \textbf{المعطيات:} في لعبة حظ، يدفع اللاعب مبلغاً قدره $m$ دينار ليرمي حجر نرد منتظم. يربح 100 دج إذا ظهر رقم زوجي، ويربح 50 دج إذا ظهر الرقم 5، ويخسر ما دفعه في باقي الحالات. ليكن $X$ الربح الصافي (الربح ناقص $m$). \\
    \textbf{السؤال:} عين قيمة $m$ حتى تكون اللعبة عادلة (أي $E(X)=0$).
    
    \item \textbf{المعطيات:} لدينا قطعة نقدية غير متزنة حيث احتمال ظهور الوجه هو ضعف احتمال ظهور الظهر. \\
    \textbf{السؤال:} احسب احتمال ظهور الوجه.
    
    \item \textbf{المعطيات:} يصيب رامي هدفه باحتمال $p = 0.8$. يرمي الرامي 5 رميات متتالية ومستقلة. \\
    \textbf{السؤال:} ما احتمال إصابة الهدف 3 مرات بالضبط؟
    
    \item \textbf{المعطيات:} نفس الرامي السابق ($p=0.8$) يرمي $n$ مرة متتالية ومستقلة. \\
    \textbf{السؤال:} ما هو الحد الأدنى لعدد الرميات $n$ حتى يكون احتمال إصابة الهدف مرة واحدة على الأقل أكبر من 0.99؟
    
    \item \textbf{المعطيات:} كيس يحوي 3 كرات حمراء و 2 خضراء. التجربة الأولى: سحب 3 كرات على التوالي بإرجاع. التجربة الثانية: سحب 3 كرات على التوالي بدون إرجاع. \\
    \textbf{السؤال:} احسب احتمال الحصول على 3 كرات حمراء في كل تجربة على حدة وقارن بين النتيجتين.
    
    \item \textbf{المعطيات:} كيس يحوي $n$ كرة بيضاء ($n \ge 2$) وكرتين سوداوين. نسحب كرتين في آن واحد وبشكل عشوائي. \\
    \textbf{السؤال:} احسب الاحتمال $P_n$ لسحب كرتين بيضاوين بدلالة $n$.
    
    \item \textbf{المعطيات:} ليكن الاحتمال المعطى بالعبارة $P_n = \frac{n(n-1)}{(n+2)(n+1)}$. \\
    \textbf{السؤال:} ادرس نهاية الاحتمال $P_n$ لما يؤول $n$ إلى $+\infty$ وفسر النتيجة المحصل عليها.
    
    \item \textbf{المعطيات:} بناءً على عبارة الاحتمال $P_n$ السابقة. \\
    \textbf{السؤال:} بين أن المتتالية $(P_n)$ متزايدة تماماً واستنتج أنها متقاربة.
    
    \item \textbf{المعطيات:} ليكن المتغير العشوائي $X$ حيث علم أن أمله الرياضي $E(X) = 2$ وتباينه $V(X) = 1.5$. ليكن المتغير الجديد $Y = 2X^2 - 1$. \\
    \textbf{السؤال:} احسب الأمل الرياضي $E(Y)$.
    
    \item \textbf{المعطيات:} كيس يحوي 3 كرات حمراء وكرة بيضاء واحدة. نسحب كرة، إذا كانت حمراء لا نرجعها ونضيف بدلها كرة بيضاء، وإذا كانت بيضاء نرجعها للكيس. ثم نقوم بسحب كرة ثانية. \\
    \textbf{السؤال:} ارسم شجرة الاحتمالات التي تنمذج هذه التجربة ثم احسب احتمال سحب كرتين بيضاوين.
    
    \item \textbf{المعطيات:} لدينا مجموعة الأرقام $\{0, 1, 2, 3, 4\}$. نريد تكوين عدد يتألف من 3 أرقام مختلفة مثنى مثنى. \\
    \textbf{السؤال:} ما هو احتمال أن يكون العدد المكون زوجياً؟
    
    \item \textbf{المعطيات:} لتكن التجربة: رمي حجر نرد منتظم مرتين متتاليتين. نُعرف الحادثة $A$: "مجموع الرقمين الظاهرين يساوي 7"، والحادثة $B$: "الرقم الظاهر في الرمية الأولى هو 3". \\
    \textbf{السؤال:} برهن أن الحادثتين $A$ و $B$ مستقلتان رياضياً.
    
    \item \textbf{المعطيات:} كيس يحوي 4 كرات بيضاء و 3 كرات سوداء. نسحب 3 كرات على التوالي وبدون إرجاع. \\
    \textbf{السؤال:} احسب احتمال الحصول على كرات ذات ألوان متناوبة.
    
    \item \textbf{المعطيات:} مريض يختار عشوائياً بين 3 عيادات طبية: A (نسبة اختيارها 50\%، ونسبة الشفاء فيها 80\%)، B (نسبة اختيارها 30\%، ونسبة الشفاء 90\%)، C (نسبة اختيارها 20\%، ونسبة الشفاء 60\%). عُلم أن المريض قد شُفي تماماً. \\
    \textbf{السؤال:} باستخدام قانون بايز، ما احتمال أنه تلقى العلاج في العيادة B؟
\end{enumerate}
